import { expect, it } from "vitest";
import { applyBookPatch, optimisticFields, pickFields, revertBookPatch } from "/home/arthur/projects/portfolio-projects/book-manager/lib/client/book-cache";
it("two failing edits same field", () => {
  let cache: any[] = [{ id: "a", status: "nao-lido", borrower: null }];
  const mutate = (patch: any) => {
    const applied = optimisticFields(patch);
    const prev = pickFields(cache[0], applied);
    cache = applyBookPatch(cache, "a", applied);
    return { applied, prev };
  };
  const c1 = mutate({ status: "lido" });
  const c2 = mutate({ status: "lendo" });
  cache = revertBookPatch(cache, "a", c1.applied, c1.prev); // edit1 fails first (queue)
  cache = revertBookPatch(cache, "a", c2.applied, c2.prev); // then edit2 fails
  expect(cache[0].status).toBe("nao-lido");
});
